preorder_idx(bp_pos)

rank_1(bp_pos) - 1 // count of 1-bits up to bp_pos, minus 1

first_child(pre_idx)

op = select_1(pre_idx + 1) B[op+1]==1 ? rank_1(op+1)-1 : None

next_sibling(pre_idx)

cp = match_pos[open_pos(pre_idx)] B[cp+1]==1 ? rank_1(cp+1)-1 : None

subtree_size(pre_idx)

op = select_1(pre_idx + 1) (match_pos[op] - op + 1) / 2

depth(pre_idx)

op = select_1(pre_idx + 1) 2 * rank_1(op) - op - 1

LCA(u, v)

l=open_pos(u), r=open_pos(v) rank_1(RMQ(l,r)) - 1 → O(1)

parent(pre_idx)

parent_map[pre_idx] // dense array // O(1), no bit navigation needed

is_leaf(pre_idx)

op = select_1(pre_idx + 1) B[op + 1] == 0
